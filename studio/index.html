    function buildTypeScriptSdk(resourceName, fields, plural, authEnabled) {
      const resTitle = resourceName.charAt(0).toUpperCase() + resourceName.slice(1);
      const pk = (fields.find(f => f.isPk) || fields[0]).name;

      return `/**
 * ${resTitle} Typed Client SDK.
 * Generated by APIGen Studio.
 * Zero external dependencies (uses native fetch).
 */

export interface ${resTitle} {
${fields.map(f => `  ${f.name}: ${f.tsType};`).join('\n')}
}

export interface ${resTitle}CreateInput {
${fields.filter(f => !f.isPk).map(f => `  ${f.name}: ${f.tsType};`).join('\n')}
}

export interface ClientConfig {
  baseUrl: string;
  apiKey?: string;
  fetch?: typeof fetch;
}

export class ${resTitle}Client {
  private baseUrl: string;
  private apiKey?: string;
  private customFetch: typeof fetch;

  constructor(config: ClientConfig) {
    this.baseUrl = config.baseUrl.replace(/\\/+$/, "");
    this.apiKey = config.apiKey;
    this.customFetch = config.fetch || globalThis.fetch;
  }

  private async request(path: string, options: RequestInit = {}): Promise {
    const headers: Record = {
      "Content-Type": "application/json",
      ...(options.headers as Record),
    };
    if (this.apiKey) {
      headers["X-API-Key"] = this.apiKey;
    }
    const res = await this.customFetch(\`\${this.baseUrl}\${path}\`, { ...options, headers });
    if (!res.ok) throw new Error(\`API Error [\${res.status}]: \${await res.text()}\`);
    return res.status === 204 ? null as T : res.json();
  }

  async health(): Promise<{ status: string; records: number }> {
    return this.request("/healthz");
  }

  async list(params: { limit?: number; offset?: number; q?: string } = {}): Promise<${resTitle}[]> {
    const query = new URLSearchParams();
    if (params.limit) query.set("limit", String(params.limit));
    if (params.offset) query.set("offset", String(params.offset));
    if (params.q) query.set("q", params.q);
    const qs = query.toString();
    return this.request(\`/${plural}\${qs ? \`?\${qs}\` : ""}\`);
  }

  async get(id: string | number): Promise<${resTitle}> {
    return this.request(\`/${plural}/\${encodeURIComponent(id)}\`);
  }

  async create(payload: ${resTitle}CreateInput): Promise<${resTitle}> {
    return this.request(\`/${plural}\`, { method: "POST", body: JSON.stringify(payload) });
  }
}
`;
    }

      state.fields = inferSchema(state.records);
      renderSchemaTable();

      if (!state.records.length) return;

      state.allRecords = generateSyntheticMocks(state.fields, state.records, state.mockCount);
      const pkField = state.fields.find(f => f.isPk) || state.fields[0];
      state.simulatedDb = {};
      state.allRecords.forEach(r => {
        state.simulatedDb[String(r[pkField.name])] = r;
      });

      const plural = state.resourceName.endsWith('s') ? state.resourceName : state.resourceName + 's';
      simEndpoint.value = `/${plural}?limit=5`;
      simRecordsTotal.textContent = `${state.allRecords.length} items stored`;

    function resetSamplePostBody() {
      const pkField = (state.fields.find(f => f.isPk) || state.fields[0])?.name;
      const sampleCreate = {};
      state.fields.filter(f => f.name !== pkField).forEach(f => {
        sampleCreate[f.name] = f.tsType === 'number' ? 99.5 : f.tsType === 'boolean' ? true : `New ${f.name}`;
      });
      simPostBody.value = JSON.stringify(sampleCreate, null, 2);
    }

    // Simulator Execution Engine (Immediate Context)

    function executeSimulation() {
      const method = state.currentMethod;
      const endpoint = simEndpoint.value.trim();
      const startTime = performance.now();

      const plural = state.resourceName.endsWith('s') ? state.resourceName : state.resourceName + 's';
      const pkField = (state.fields.find(f => f.isPk) || state.fields[0])?.name;

      let status = 200;
      let response = {};

      if (endpoint === '/healthz') {
        response = { status: "healthy", service: `${state.resourceName}-api`, records_count: Object.keys(state.simulatedDb).length };
      } else if (endpoint.startsWith(`/${plural}`)) {
        if (method === 'GET') {
          const singleMatch = endpoint.match(new RegExp(`^/${plural}/([^?]+)`));
          if (singleMatch) {
            const id = singleMatch[1];
            if (state.simulatedDb[id]) {
              response = state.simulatedDb[id];
            } else {
              status = 404;
              response = { detail: `${state.resourceName} not found` };
            }
          } else {
            let items = Object.values(state.simulatedDb);
            const urlParams = new URLSearchParams(endpoint.split('?')[1] || '');
            const limit = parseInt(urlParams.get('limit'), 10) || 10;
            const offset = parseInt(urlParams.get('offset'), 10) || 0;
            const q = urlParams.get('q');

            if (q) {
              const query = q.toLowerCase();
              items = items.filter(it => Object.values(it).some(v => String(v).toLowerCase().includes(query)));
            }

            response = {
              items: items.slice(offset, offset + limit),
              pagination: {
                total: items.length,
                limit,
                offset
              }
            };
          }
        } else if (method === 'POST') {
          try {
            const payload = JSON.parse(simPostBody.value);
            const newId = `ID_${Object.keys(state.simulatedDb).length + 101}`;
            payload[pkField] = newId;
            state.simulatedDb[newId] = payload;
            status = 201;
            response = payload;
            simRecordsTotal.textContent = `${Object.keys(state.simulatedDb).length} items stored`;
          } catch (e) {
            status = 422;
            response = { detail: "Malformed JSON payload in request body" };
          }
        } else if (method === 'DELETE') {
          const singleMatch = endpoint.match(new RegExp(`^/${plural}/([^?]+)`));
          if (singleMatch && state.simulatedDb[singleMatch[1]]) {
            delete state.simulatedDb[singleMatch[1]];
            status = 204;
            response = { status: "deleted" };
            simRecordsTotal.textContent = `${Object.keys(state.simulatedDb).length} items stored`;
          } else {
            status = 404;
            response = { detail: "Item not found" };
          }
        }
      } else {
        status = 404;
        response = { detail: "Path not found in OpenAPI spec" };
      }

      const duration = Math.round(performance.now() - startTime + Math.random() * 3);
      simLatencyBadge.textContent = `${status} ${status === 201 ? 'CREATED' : status === 204 ? 'NO CONTENT' : status === 404 ? 'NOT FOUND' : 'OK'} (${duration}ms)`;
      simLatencyBadge.className = status >= 400 ? 'text-rose-400 font-mono font-semibold' : 'text-emerald-400 font-mono font-semibold';
      simOutput.textContent = JSON.stringify(response, null, 2);
    }
